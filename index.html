function changeSlide(direction) { const target = currentSlide + direction; if (target >= 1 && target <= totalSlides) { goToSlide(target); } } function goToSlide(slideNum) { document.getElementById(`slide-${currentSlide}`).classList.add('hidden'); document.getElementById(`slide-${currentSlide}`).classList.remove('flex'); currentSlide = slideNum; const nextSlideEl = document.getElementById(`slide-${currentSlide}`); nextSlideEl.classList.remove('hidden'); nextSlideEl.classList.add('flex', 'fade-enter-active'); updateSlideView(); renderDots(); lucide.createIcons(); } function updateSlideView() { document.getElementById('slide-counter').innerText = `Pantalla ${currentSlide} de ${totalSlides}`; document.getElementById('slide-select').value = currentSlide; const pct = (currentSlide / totalSlides) * 100; document.getElementById('progress-bar').style.width = `${pct}%`; document.getElementById('btn-prev').disabled = (currentSlide === 1); if (currentSlide === totalSlides) { document.getElementById('btn-next').innerHTML = `Finalizar `; } else { document.getElementById('btn-next').innerHTML = `Siguiente `; } }
